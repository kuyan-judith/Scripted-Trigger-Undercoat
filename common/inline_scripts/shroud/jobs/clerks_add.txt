# Default Clerk (Clerk)
triggered_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_gestalt = no
			is_tankbound_empire = no
		}
	}
	modifier = {
		job_clerk_add = $AMOUNT$
	}
}
triggered_desc = {
	trigger = {
		exists = owner
		owner = {
			is_gestalt = no
			is_tankbound_empire = no
		}
	}
	text = job_clerk_effect_desc
}
