# Default Priest (Priest)

	potential = {
		exists = owner
		owner = {
			is_megacorp = no

				is_spiritualist = yes

			}
			is_superstitious_beliefs_empire = no
			is_cybernetic_creed_empire = no # undercoat: moved to scripted trigger
		}
	}

		job_bureaucrat_add = $AMOUNT$
	}
