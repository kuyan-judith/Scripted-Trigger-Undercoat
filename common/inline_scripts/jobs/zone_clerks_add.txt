### Regular Empires

# Planets
triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_gestalt = no
			is_tankbound_empire = no
		}
		planet_is_habitat_equivalent = no
		planet_is_ecu_equivalent = no
		planet_is_ring_world_equivalent = no
	}
	job_clerk_add = $AMOUNT$
}

triggered_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
			has_active_tradition = tr_virtuality_4
			is_tankbound_empire = no
		}
		planet_is_habitat_equivalent = no
		planet_is_ecu_equivalent = no
		planet_is_ring_world_equivalent = no
	}
	job_clerk_add = $AMOUNT$
	mult = 0.5
}

### Tankbound Clerk (Production Overseer)
# Planets
triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = { is_tankbound_empire = yes }
		NOR = {
			is_habitat = yes # undercoat: moved to scripted trigger
			is_city_world = yes # undercoat: moved to scripted trigger
			is_machine_world = yes # undercoat: moved to scripted trigger
			is_hive_world = yes # undercoat: moved to scripted trigger
			is_unshattered_ring_habitable_segment = yes # undercoat: moved to scripted trigger
		}
	}
	job_production_overseer_add = $AMOUNT$
}
